```sql tijden
select avg(behandeltijd) as gemiddeldeBehandeltijdMin, 
       avg(wachttijd) as gemiddeldeWachttijdMin, 
       median(wachttijd) as mediaanWachttijdMin 
  from fct_gebeurtenis
```

```sql aankomst_vertrek

  select date_trunc('month', fg.startTijdstip) as maand,
         'Aankomst' as direction,
         count(fg.patientId) as count
    from fct_gebeurtenis fg
    left join dim_patient p on fg.patientId = p.id
    left join dim_aandoening a on p.aandoeningId = a.id
    left join dim_behandelkamer b on fg.behandelkamerId = b.id
    where b.behandelkamerType = 'Receptie'
    group by all

union all 

    select date_trunc('month', fg."eindTijdstip") as maand,
       'Vertrek' as direction,
       count(fg.patientId) * -1 as count
    from fct_gebeurtenis fg
    left join dim_patient p on fg.patientId = p.id
    left join dim_aandoening a on p.aandoeningId = a.id
    left join dim_behandelkamer b on fg.behandelkamerId = b.id
    where b.behandelkamerType in ('Kliniek', 'Polikliniek')
    group by all


```

```sql routes_aantallen
with volume as (
  select r.Diagnose, r.Behandeling, count(*) as patienten
  from dim_patient p
  join dim_aandoeningRoutes r on p.aandoeningId = r.AandoeningId
  group by all
)
select 'Reception' as source, Diagnose as target, sum(patienten) as value
from volume group by all
union all
select Diagnose, Behandeling, sum(patienten)
from volume group by all
```

```sql aankomst_aandoening
  select date_trunc('month', fg.startTijdstip) as maand,
         a.aandoeningNaam,
         count(fg.patientId) as aankomsten
from fct_gebeurtenis fg
  left join dim_patient p on fg.patientId = p.id
  left join dim_aandoening a on p.aandoeningId = a.id
  left join dim_behandelkamer b on fg.behandelkamerId = b.id
  inner join (select aandoeningId, count(*) as aantal
              from dim_patient
              left join dim_aandoening ON (dim_aandoening.id = dim_patient."aandoeningId")
              group by aandoeningId
              order by aantal desc
              limit 5) top5 on a.id = top5.aandoeningId
where b.behandelkamerType = 'Receptie'
group by all
order by maand, a.aandoeningNaam;
```

```sql capaciteit
with toestand as (
  -- elke snapshot geldt tot de volgende snapshot van dezelfde kamer
  select  behandelkamerId, 
          tijdstip, 
          capaciteit, 
          patientenInBehandeling, 
          patientenInWachtrij,
          (epoch_ms(lead(tijdstip) over (partition by simulatieRun, behandelkamerId order by tijdstip, id))
          - epoch_ms(tijdstip)) / 60000.0                                       as duurMinuten
  from fct_behandelkamer
)
select  date_trunc('month', t.tijdstip)::date                                   as maand,
        b.behandelkamerNaam,
        b.behandelkamerType,
        sum(t.capaciteit * t.duurMinuten) / 60                                  as beschikbareUren,
        sum(t.patientenInBehandeling * t.duurMinuten) / 60                      as bezetteUren,
        sum(t.patientenInBehandeling * t.duurMinuten)
        / nullif(sum(t.capaciteit * t.duurMinuten), 0)                          as benuttingsgraad,
        
        -- gemiddelden alleen over openingstijd (capaciteit > 0)
        sum(t.patientenInBehandeling * t.duurMinuten) filter (where t.capaciteit > 0)
          / nullif(sum(t.duurMinuten) filter (where t.capaciteit > 0), 0)       as gemBezetteKamers,
        sum(t.patientenInWachtrij * t.duurMinuten) filter (where t.capaciteit > 0)
          / nullif(sum(t.duurMinuten) filter (where t.capaciteit > 0), 0)       as gemWachtrij,
        max(t.patientenInWachtrij)                                              as maxWachtrij,
        max(t.capaciteit)                                                       as maxCapaciteit
 from toestand t
 join dim_behandelkamer b on t.behandelkamerId = b.id

where t.duurMinuten is not null
  and b.behandelkamerNaam in ${inputs.kamer.value}
  and b.behandelkamerType in ${inputs.kamerType.value}

group by all

order by 
maand, 
b.behandelkamerNaam


```
