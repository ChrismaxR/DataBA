```sql aandoeningDropdown
select distinct aandoeningOmschrijving
from hospital.aankomstOntslag

```


```sql aankomstAgg

select aankomstMaand, 
       aankomstMaandLabel,
       ontslagen,
       sum(patienten) patienten 
  from aankomstOntslag

where aandoeningOmschrijving in ${inputs.aandoening.value}
  and ontslagen = true

group by 
  aankomstMaand, 
  aankomstMaandLabel, 
  ontslagen

```


```sql vertrekAgg
select aankomstMaand, 
       aankomstMaandLabel,
       sum(patienten) patienten 
  from aankomstOntslag

where aandoeningOmschrijving in ${inputs.aandoening.value}
  and ontslagen = false

group by 
  aankomstMaand, 
  aankomstMaandLabel

```

```sql aankomst
select *
  from aankomstOntslag

where aandoeningOmschrijving in ${inputs.aandoening.value}

```
